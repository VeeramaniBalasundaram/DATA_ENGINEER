-----------------------------------------------------------CTE WITH WHERE CONDITION LOGICS--------------------------------------------------------------------------
 WITH FIRST_DATA AS 
 (
	SELECT	
		UPPER(FIRSTNAME) AS NEW_NAME,
		UPPER(LASTNAME) AS LAST_NAME,
		UPPER(CONCAT(FIRSTNAME , LASTNAME)) AS FULL_NAME,
		CASE 
			WHEN GENDER = 'F' THEN 'FEMALE'
			WHEN GENDER = 'M' THEN 'MALE'
		END AS NEW_GENDER_NAME,
		CASE 
			WHEN SALARY < 70000 THEN 'LOW'
			WHEN SALARY BETWEEN 70000  AND 100000 THEN 'MEDIUM'
			ELSE 'HIGH'
		END AS NEW_SALARY,
		CASE 
			WHEN BONUS < 20000 THEN 'LOW'
			WHEN BONUS BETWEEN 20000  AND 40000 THEN 'MEDIUM'
			ELSE 'HIGH'
		END AS NEW_BONUS,
		YEAR(HIREDATE) AS ONLY_YEAR,
		LOCATION,
		AGE 
	FROM EMPLOYEES_SUBQUERY
	WHERE Location = 'CHENNAI'
)
SELECT * FROM FIRST_DATA  WHERE NEW_SALARY = 'MEDIUM';

		SALARY * 2 AS MULTIPLE_SALARY,

)

--------------------------------------------------------CTE + JOIN-----------------------------------------
--------------------------------------------------------Very important for Data Engineering.-----------------------------------------
WITH JOIN_TESTING AS 
	(
		SELECT * FROM EMPLOYEES 
		WHERE STATUS = 'ACTIVE' 
	)
					SELECT 
					UPPER(CONCAT_WS('_' , EP.FIRSTNAME , EP.LASTNAME)) AS NEW_NAME,
					CHOOSE(CHARINDEX(EP.GENDER , 'MF') , 'MALE' , 'FEMALE') AS NEW_GENDER,
					YEAR(EP.DATEOFBIRTH) AS YEAR_ONLY,
					YEAR(EP.HIREDATE) AS YEAR_HIREDATE,
					EP.SALARY,
					EP.BONUS,
					EP.STATUS,
					DP.DEPARTMENTNAME,
					DP.LOCATION,
					YEAR(DP.CREATEDDATE) AS YEAR_ONLY,
					PS.PROJECTNAME,
					PS.BUDGET
					FROM JOIN_TESTING AS EP
						INNER JOIN DEPARTMENTS AS DP
						ON EP.DEPARTMENTID = DP.DEPARTMENTID
						INNER JOIN PROJECTS AS PS
						ON PS.DEPARTMENTID = EP.DEPARTMENTID
					WHERE BONUS IS NOT NULL

WITH EMPLOYESS_DATA AS 
	(
		SELECT * FROM EMPLOYEES 
		WHERE STATUS IN  ('ACTIVE' , 'INACTIVE')
	),
DEPARTMENTS_DATA AS 
	(
		SELECT * FROM DEPARTMENTS 
		WHERE LOCATION IN ( 'CHENNAI' , 'PUNE' , 'MUMBAI' , 'DELHI')
	),
PROJECTS_DATA AS 
	(
		SELECT * FROM PROJECTS 
		WHERE PROJECTSTATUS = 'ACTIVE'
	)
				
				SELECT 
					FIRSTNAME,
					LASTNAME,
					SALARY,
					GENDER
					FROM EMPLOYESS_DATA AS E 
					INNER JOIN DEPARTMENTS_DATA AS D
					ON E.DEPARTMENTID = D.DEPARTMENTID
					INNER JOIN PROJECTS_DATA AS P
					ON E.DEPARTMENTID = P.DEPARTMENTID ;

WITH MULTIPLE_TESTING_IN AS 
(	
	SELECT 
			DEPARTMENTID,
			UPPER(CONCAT_WS('_' , FIRSTNAME , LASTNAME)) AS NEW_NAME,
			CHOOSE(CHARINDEX(GENDER , 'MF') , 'MALE' , 'FEMALE') AS NEW_GENDER,
			IIF(GENDER = 'M' , 'MALE' , 'FEMALE') AS IF_LOGIC_GENDER,
			IIF(SALARY > 100000 , 'OK' , 'NOT_OK') AS NEW_SALARY,
			IIF(BONUS >= 50000 , 'BONUS_OK' , 'BONUS_NOT_OK') AS BONUS_VALUE,
			YEAR(HIREDATE) AS YEAR_ONLY,
			DATEDIFF(YEAR , HIREDATE , GETDATE()) AS DIFFERENCE_DATE,
			DATEADD(YEAR , 10 , GETDATE()) AS ADDED_DATE,
			LOCATION
			FROM EMPLOYEES_SUBQUERY
)
	SELECT  * FROM MULTIPLE_TESTING_IN 
	WHERE DEPARTMENTID IN --------------------------MULTIPLE SUB QUERY CTE  , THIS QUERY IS JOINING THE ABOVE QUERY AND BELOW QUERY 

(	
	SELECT DEPARTMENTID FROM DEPARTMENTS_SUBQUERY
	WHERE LOCATION = 'CHENNAI'
)
