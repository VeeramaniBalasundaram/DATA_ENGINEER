----------------------------------------------CTE ---- COMMON TABLE EXPRESSION ----------------------------------------------
----- WHAT IS CTE --- IT IS COMMON TABLE EXPRESSION ------
----A CTE is a named temporary result set that you define using WITH and then use in the query immediately following it.------------------------
 
 WITH FIRST_DATA AS 
 (
	SELECT	
		UPPER(FIRSTNAME) AS NEW_NAME,
		UPPER(LASTNAME) AS LAST_NAME,
		UPPER(CONCAT(FIRSTNAME , LASTNAME)) AS FULL_NAME,
		CASE 
			WHEN GENDER = 'F' THEN 'FEMALE'
			WHEN GENDER = 'M' THEN 'MALE'
		END AS NEW_GENDER_NAME,
		CASE 
			WHEN SALARY < 70000 THEN 'LOW'
			WHEN SALARY BETWEEN 70000  AND 100000 THEN 'MEDIUM'
			ELSE 'HIGH'
		END AS NEW_SALARY,
		CASE 
			WHEN BONUS < 20000 THEN 'LOW'
			WHEN BONUS BETWEEN 20000  AND 40000 THEN 'MEDIUM'
			ELSE 'HIGH'
		END AS NEW_BONUS,
		YEAR(HIREDATE) AS ONLY_YEAR,
		LOCATION,
		AGE 
	FROM EMPLOYEES_SUBQUERY
)
SELECT * FROM FIRST_DATA  WHERE NEW_SALARY = 'MEDIUM'

WITH SCALAR_CTE_DATA_SELECT_LOCATION AS 
(
			SELECT * FROM EMPLOYEES_SUBQUERY
)
			SELECT 
					UPPER(CONCAT_WS('_' , FIRSTNAME , LASTNAME)) AS NEW_NAME,
					CHOOSE(CHARINDEX(GENDER , 'MF') , 'MALE' , 'FEMALE') AS NEW_GENDER,
					IIF(GENDER = 'M' , 'MALE' , 'FEMALE') AS IF_LOGIC_GENDER,
					IIF(SALARY > 100000 , 'OK' , 'NOT_OK') AS NEW_SALARY,
					IIF(BONUS >= 50000 , 'BONUS_OK' , 'BONUS_NOT_OK') AS BONUS_VALUE,
					YEAR(HIREDATE) AS YEAR_ONLY,
					DATEDIFF(YEAR , HIREDATE , GETDATE()) AS DIFFERENCE_DATE,
					DATEADD(YEAR , 10 , GETDATE()) AS ADDED_DATE,
						(
							SELECT AVG(SALARY)  -------------- SCALAR SUBQUERY WITH SELECTION LOCTION ONLY ONE DATA CAN ENTER INSIDE THE QUERY CALLED AS A SCALAR SUB QUERY  
							FROM EMPLOYEES_SUBQUERY AS SCALAR_SUB_QUERY_AVG
						),
						(
							SELECT MIN(SALARY)  -------------- SCALAR SUBQUERY WITH SELECTION LOCTION ONLY ONE DATA CAN ENTER INSIDE THE QUERY CALLED AS A SCALAR SUB QUERY  
							FROM EMPLOYEES_SUBQUERY AS SCALAR_SUB_QUERY_MIN
						),
							(
							SELECT MAX(SALARY)  -------------- SCALAR SUBQUERY WITH SELECTION LOCTION ONLY ONE DATA CAN ENTER INSIDE THE QUERY CALLED AS A SCALAR SUB QUERY  
							FROM EMPLOYEES_SUBQUERY AS SCALAR_SUB_QUERY_MAX
						),
						(
							SELECT SUM(SALARY)  -------------- SCALAR SUBQUERY WITH SELECTION LOCTION ONLY ONE DATA CAN ENTER INSIDE THE QUERY CALLED AS A SCALAR SUB QUERY  
							FROM EMPLOYEES_SUBQUERY AS SCALAR_SUB_QUERY_SUM
						),
					LOCATION
					FROM SCALAR_CTE_DATA_SELECT_LOCATION;

					MIN(SALARY) AS MIN_SALARY,
