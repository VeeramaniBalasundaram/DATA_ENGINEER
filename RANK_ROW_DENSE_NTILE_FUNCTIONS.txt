WITH MULTIPLE_CASE_WINDOWS_FUNCTION AS
(
		SELECT
			UPPER(FIRSTNAME) AS NEW_NAME,
			UPPER(LASTNAME) AS LAST_NEW_NAME,
			CHOOSE(CHARINDEX(GENDER , 'MF' ),'MALE', 'FEMALE') AS NEW_GENDER,
			IIF(GENDER =  'M' ,'MALE' , 'FEMALE' ) AS IIF_GENDER,
			CASE 
				WHEN GENDER  = 'M' THEN 'MALE'---------------------------------CASE WITH CTE AND WINDOWS_FUNCTION 
				ELSE 'FEMALE'
			END AS CASE_GENDER ,
			CASE 
				WHEN SALARY < 50000 THEN 'LOW_SALARY'
				WHEN SALARY BETWEEN 100000 AND 200000 THEN 'MEDIUM' ---------------------------------CASE WITH CTE AND WINDOWS_FUNCTION 
				ELSE'HIGH'
			END AS NEW_SALARY,
			CASE 
				WHEN BONUS < 10000 THEN 'LOW'
				WHEN BONUS BETWEEN 20000 AND 50000 THEN 'MEDIUM'---------------------------------CASE WITH CTE AND WINDOWS_FUNCTION 
				ELSE'HIGH'
			END AS NEW_BONUS, 
			YEAR(HIREDATE) - YEAR(GETDATE()) AS DIFFERENCE_DATE,
			LOCATION,
			ROW_NUMBER () OVER (PARTITION BY LOCATION ORDER BY SALARY DESC) AS UNIQUE_ROWS
			FROM EMPLOYEES_WINDOWS
)
SELECT * FROM MULTIPLE_CASE_WINDOWS_FUNCTION
WHERE UNIQUE_ROWS  <= 10
AND LOCATION = 'CHENNAI'

---------------------------------------Gives same rank for ties, but leaves gaps after the tie.------------------------------------

------RANKING BY EMPLOYEE-------------------------
SELECT 
	EMPLOYEEID,
	SUM(SALARY) AS TOTAL_SALARY,
	RANK() OVER(ORDER BY SUM(SALARY) DESC) AS RANK_FUNCTION,-------------------------Rank employees according to their total sales.
	RANK() OVER(ORDER BY AVG (SALARY) DESC) AS RANK_AVG,------------------------------Rank employees according to their AVG sales.
	RANK() OVER(ORDER BY MIN (SALARY) DESC) AS RANK_MIN,----------------------------Rank employees according to their MIN sales.
	RANK() OVER(ORDER BY MAX (SALARY) DESC) AS RANK_MAX--------------------------------Rank employees according to their MAX sales.

	GROUP BY EMPLOYEEID

			WHEN LOACTION_WISE_DATA = 3 THEN 'THIRD_BUCKET'
			WHEN LOACTION_WISE_DATA = 4 THEN 'FOURTH_BUCKET'
		END AS BUCKET_WINDOWS
		FROM NTILE_WINDOW_FUNCTION 
),
NTILE_WINDOW_FUNCTION_FINAL_LOCATION AS 

(
	SELECT * FROM NTILE_WINDOW_FUNCTION_FINAL
	WHERE BUCKET_WINDOWS = 'FIRST_BUCKET'
)
SELECT * FROM NTILE_WINDOW_FUNCTION_FINAL_LOCATION
WHERE LOCATION = 'CHENNAI'
ORDER BY SALARY DESC ;

WITH NTILE_WINDOW_FUNCTION AS 
(
		SELECT 
		LOCATION,
		SALARY,
		UPPER(CONCAT(FIRSTNAME, LASTNAME)) AS NEW_NAME,
		NTILE(4) OVER (PARTITION BY LOCATION ORDER BY EXPERIENCEYEARS DESC) AS LOACTION_WISE_DATA -------- 2 GROUPS MEANS2 BUCKET (TOTAL ROWS 100 SO :50 + 50 )------------- FROM 51 TO ANOTHER BUCKET LEVEL 
		FROM WINDOWS_FUNCTION

			LOACTION_WISE_DATA,

			FROM NTILE_WINDOW_FUNCTION

		EMPLOYEEID,
		SUM(SALARY) AS NEW_SALARY                         
		FROM WINDOWS_FUNCTION 
		GROUP BY EMPLOYEEID 
),
NTILE_WINDOW_FUNCTION_FINAL AS 
(
		SELECT
			EMPLOYEEID,
			NTILE(4) OVER (ORDER BY NEW_SALARY  DESC) AS LOACTION_WISE_DATA
			FROM NTILE_WINDOW_FUNCTION
