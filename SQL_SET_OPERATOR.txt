	SELECT  * ,
	CASE STATUS 
	WHEN 'ACTIVE' THEN 'OK'
	WHEN 'ON_LEAVE' THEN 'NOTOK'
	WHEN 'RESIGNED' THEN 'NULL'
	WHEN 'INACTIVE' THEN 'DELETED'
	ELSE STATUS
	END 
	FROM EMP_DATA
	WHERE CITY = 'CHENNAI'

	
	SELECT 
	DEPARTMENT ,
	CASE DEPARTMENT 
	WHEN  'IT' THEN 'INFORMATION'
	WHEN  'QA' THEN 'QUALITY_ACESS'
	WHEN  'R&D' THEN 'RESERCH_AND_DEVELOPEMENT'
	ELSE DEPARTMENT
	END AS UPDATED_COLUMN,
	GENDER,
	CASE GENDER
	WHEN 'F' THEN 'FEMALE'
	WHEN 'M' THEN 'MALE'
	END AS UPDATED_NAME,
	STATUS,
	CASE STATUS
	WHEN 'ACTIVE' THEN '0K'
	ELSE 'NOT OK'
	END AS UPDATED_STATUS,
	SALARY,
	SUM(SALARY) AS NEW_SALARY,
	AVG(SALARY) AS NEW_AVG_SALARY,
	MIN(SALARY) AS MIN_SALARY,
	MAX(SALARY) AS MAX_SALARY,
	COUNT(*) AS TOTAL_EMPLOYEES
	FROM EMP_DATA
	WHERE DEPARTMENT  IN ('IT' , 'R&D' , 'ADMIN' , 'SUPPORT' , 'PRODUCTION')
	GROUP BY DEPARTMENT , GENDER , STATUS , SALARY 
	HAVING SUM(SALARY) > 190000
	ORDER BY SALARY DESC 
	
	SELECT 
	* ,
	CASE
    WHEN Salary BETWEEN 180000 AND 190000 THEN 'GOOD_SALARY'
    WHEN Salary BETWEEN 170000 AND 180000 THEN 'NORMAL_SALARY'
    WHEN Salary BETWEEN 120000 AND 170000 THEN 'ACCEPTABLE'
    WHEN Salary < 120000 THEN 'BAD'
    ELSE 'UNKNOWN'
	END AS Salary_Expectation
	FROM EMP_DATA
	WHERE DEPARTMENT = 'IT'  AND CITY = 'CHENNAI'
