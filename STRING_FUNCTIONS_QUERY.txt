WITH DIFFERENCE_CTE AS
(
	SELECT 
		GETDATE() AS DATE_TYPE, ------------THIS IS TODAY DATE AND WITH MINUTS ANS SECONDS 
		GETUTCDATE() AS UTC_DATE, ------------- THIS IS UTC DATE AND TIME ( THAT MENS FROM TODAY - 5:30 HOURS DIFFEREMCE)
		SYSDATETIME () PC_DATE_TIME --------- THIS ALSO TODAY DATE WITH GOOD ACCURACY  
		FROM DATE_TIME_FUNCTION
) 
	SELECT  
	PC_DATE_TIME,
	DATEDIFF (HOUR , DATE_TYPE , UTC_DATE) AS DIFFERNCE
	FROM DIFFERENCE_CTE

------------------THIS IS DATE ADDED POSITIVE VALUES --------------------------DATEADD(
 
 /* WHAT?       → DAY / MONTH / YEAR
    HOW MUCH?   → positive / negative number                                } THIS IS FOR SYNTAX OF DATE ADD FUNCTION 
    FROM WHERE? → date/column
  */

	SELECT 
		CAST(GETDATE() AS DATE)  AS TODAY_DATE,
		CAST(DATEADD(YEAR , 10 , GETDATE()) AS DATE)  AS DIFFERNCE_AGE,
		CAST(DATEADD(MONTH , 10 , GETDATE()) AS DATE) AS MONTH_DIFFERENCE,
		CAST(DATEADD(DAY , 10 , GETDATE()) AS DATE) AS DAY_DIFFERENCE,
		DATEADD(DAYOFYEAR , 10 , GETDATE()) AS DIFFERNCE_AGE
		FROM DATE_TIME_FUNCTION
------------------THIS IS DATE ADDED NEGATIVE VALUES --------------------------
	SELECT 
		CAST(GETDATE() AS DATE)  AS TODAY_DATE,
		CAST(DATEADD(YEAR , -1 , GETDATE()) AS DATE)  AS DIFFERNCE_AGE,
		CAST(DATEADD(MONTH , -1 , GETDATE()) AS DATE) AS MONTH_DIFFERENCE,
		CAST(DATEADD(DAY , -1 , GETDATE()) AS DATE) AS DAY_DIFFERENCE,
		DATEADD(DAYOFYEAR , -1 , GETDATE()) AS DIFFERNCE_AGE
		FROM DATE_TIME_FUNCTION
