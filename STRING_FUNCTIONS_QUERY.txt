			MONTH(HIREDATE) AS HIRE_MONTH, ----------- THIS IS MONTH FUNCTION

				WHEN MONTH(HIREDATE) IN ( 4 , 5 , 6) THEN 'Q2'

				ELSE 'Q4'

			FROM DATE_TIME_FUNCTION

			/*
					DAY()
						→ Returns numeric day.
					MONTH()
						→ Returns numeric month.
		
					YEAR()
						→ Returns numeric year.
					DATEPART()
						→ Returns a numeric date part.
					DATENAME()
						→ Returns a text date part.
					------------------ FINAL REMEMBER --------------------------
					DAY(date)       → 29
					MONTH(date)     → 8
					YEAR(date)      → 2026
					DATEPART(MONTH, date) → 8

		| ISNULL   | ISNULL(expression, replacement)     | Replace NULL with a value |
		| COALESCE | COALESCE(value1, value2, ...)       | Return the first non-NULL value |
		| NULLIF   | NULLIF(expression1, expression2)    | Return NULL when two values are equal |

	*/
------------------------------------------------| ISNULL   | ISNULL(expression, replacement)     | Replace NULL with a value |------------------------------

	SELECT 
		ISNULL(FIRSTNAME , 'NAME NOT AVAILABLE'), ---------------- THIS IS USED TO TO REMOVE THE NULL AND PUT THE VALUE WHICH IS MENTIONED 
		CASE 
			WHEN FIRSTNAME IS NULL THEN '0'
			ELSE FIRSTNAME
		END,
		ISNULL(SALARY , 0) AS UPDATED_VERSION, ---------------- THIS IS USED TO TO REMOVE THE NULL AND PUT THE VALUE WHICH IS MENTIONED 
		ISNULL(SALARY , 0) + ISNULL(BONUS , 0) AS TOTAL_INCOME , ---------------- THIS IS USED TO TO REMOVE THE NULL AND PUT THE VALUE WHICH IS MENTIONED 
		CASE 
			WHEN SALARY IS NULL THEN '0'  --------- THIS IS CASE SAME AS ISNULL CONDITION BOTH WILL GIVE THE SAME OUTPUT 
			ELSE SALARY
		END AS NEW_SALARY,
		CASE 
			WHEN BONUS IS NULL THEN '0'
			ELSE BONUS
		END AS NEW_BONUS
		FROM NULL_FUNCTIONS_PRACTICE

-----------------------------------------------------CONVERT() — Simple Understanding----------------------------------------
-----------------------------------------------------CONVERT() changes one data type into another data type.-----------------
--------------------------------------------------------CONVERT(data_type, expression)---------------------------------------

						/*
						4. Where do we use CONVERT?
						Use CONVERT() when you need SQL Server-specific conversion, especially date/time formatting with STYLE.
						CONVERT()  → datatype conversion + date STYLE ⭐⭐⭐⭐⭐
						*/

			SELECT 
			CONVERT(VARCHAR(30), DATEOFBIRTH, 101) AS STYLE_101,
			CONVERT(VARCHAR(30), DATEOFBIRTH, 102) AS STYLE_102,
			CONVERT(VARCHAR(30), DATEOFBIRTH, 103) AS STYLE_103,
			CONVERT(VARCHAR(30), DATEOFBIRTH, 104) AS STYLE_104,
			CONVERT(VARCHAR(30), DATEOFBIRTH, 105) AS STYLE_105,
			CONVERT(VARCHAR(30), DATEOFBIRTH, 106) AS STYLE_106,

			CONVERT(VARCHAR(30), DATEOFBIRTH, 112) AS STYLE_112,
			CONVERT(VARCHAR(30), DATEOFBIRTH, 120) AS STYLE_120,
			CONVERT(VARCHAR(30), DATEOFBIRTH, 121) AS STYLE_121
			FROM DATE_TIME_FUNCTION;

	/*

							   CONVERT STYLE
								   ↓
					┌──────────────┴──────────────┐
					↓                             ↓
			DATE/DATETIME → VARCHAR        VARCHAR → DATE/DATETIME
					↓                             ↓
			   FORMAT IT                    READ/INTERPRET IT
					↓                             ↓
			  103 = DD/MM/YYYY              103 = DD/MM/YYYY
			  112 = YYYYMMDD                112 = YYYYMMDD
			  120 = YYYY-MM-DD HH:MI:SS     120 = YYYY-MM-DD HH:MI:SS
  
  */
