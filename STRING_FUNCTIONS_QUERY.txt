	SELECT 
		UPPER(FIRSTNAME) AS UPPER_NAME, ------------- TO CHANGE THE ALL VALUES UPPER 
		LOWER(LASTNAME) AS LOWER_NAME,  ------------- TO CHANGE THE ALL VALUES LOWER 
		UPPER(CONCAT(FIRSTNAME , LASTNAME)) AS FULL_NAME, ----------UPPER + CONCAT USED FOR TO MERGE THE BOTH COLUMN VALUES 
		LEN(FULLNAMERAW) AS TOTAL_LENGTH, ----------- TO CHECK THE VALUE OF LENGTH 
		DATALENGTH(FULLNAMERAW) AS TOTAL_DATA_AS_BYTES, ----------- TO CHECK THE TOTAL BYTES VALUES FOR THAT COLUMN
		TRIM(DEPARTMENT) AS SPACE_REMOVED_BOTHSIDES,----------- TO REMOVE THE EXTRA SPACES FROM THAT COLUMN 
		LTRIM(DEPARTMENT) AS LEFT_SIDE_SPACE_REMOCED,------------- REMOVE THE LEFT SIDE VALUE SPACES 
		RTRIM(DEPARTMENT) AS RIGHT_SIDE_SPACE_REMOVED,------------- REMOCE THE RIGHT SIDE VALUE SPACES 
		CONCAT(DEPARTMENT , LOCATION) AS DEPT_LOCATION,--------- TO ADD BOTH COLUMN VALUSE IN SINGLE ROWS

		RIGHT(EMPLOYEECODE , 4) AS RIGHT_DATA, ------- RETURN THE LAST 4 DIGIT OF VALUE 
		SUBSTRING(PHONENUMBER , 5, 10) AS PHONE_NUMBER_NEW ,------------EXTRACT THE STRING STARTING POIT POSIION TO DECICED END POINT
		CHARINDEX ('HN' , FULLNAMERAW,1) AS STARTING_POISIITON ,-----Finds the starting position of a specified string within another string.
		PATINDEX('%HN%' , FULLNAMERAW) AS NEW_PATINDEX,------------------Finds the position where HN occurs in FULLNAMERAW.
		PATINDEX('%H_N%' , FULLNAMERAW) AS NEW_PATINDEX1,---------------- Finds the position of a pattern H + any one character + N in FULLNAMERAW.
		PATINDEX('%[A-D]%' , FULLNAMERAW) AS NEW_PATINDEX2,----------------Finds the position of the first character from A through S in FULLNAMERAW.
		REPLACE (EMAIL , '@' , '__') AS NEW_STRING,
		REPLACE(FULLNAMERAW, ' ', '') AS EMPTY_SPACE_REMOVED,
		REPLACE(REPLACE(REPLACE(PHONENUMBER , '+' , '') , '-' , '') , '91' , '') AS DOUBLE_REPLCE,
		TRANSLATE(PHONENUMBER , '+' ,'/')AS TRIPLE_TRANSLATE,
		REVERSE(EMPLOYEECODE) AS REVERSE_TOTALY,
		REVERSE(FIRSTNAME) AS REVERSE_NAME,
		REPLICATE (GENDER , 3) AS THREE_TIMES_GENDER,
		FIRSTNAME + SPACE(2) + LASTNAME ,
		STUFF(MIXEDCASETEXT , 3 , 1, GENDER) 
		FROM STRING_FUNCTIONS
